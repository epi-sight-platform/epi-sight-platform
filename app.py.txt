# --- HELPER FUNCTIONS (Cleaning, Reporting, PDF) ---

def extract_text_from_pdf(file):
    # Function to extract text from a PDF file
    reader = PdfReader(file)
    text = ""
    for page in reader.pages:
        text += page.extract_text()
    return text

def smart_clean_data(df):
    # Automated data cleaning protocol
    report = []
    
    # 1. Clean Headers (Snake Case)
    old_cols = list(df.columns)
    df.columns = [c.strip().lower().replace(' ', '_').replace('-', '_') for c in df.columns]
    if list(df.columns) != old_cols:
        report.append("✅ Standardized column headers (snake_case).")
        
    # 2. Remove Duplicates
    duplicates = df.duplicated().sum()
    if duplicates > 0:
        df = df.drop_duplicates()
        report.append(f"✅ Removed {duplicates} duplicate rows.")
    
    # 3. Handle Missing Values
    num_cols = df.select_dtypes(include=['number']).columns
    if len(num_cols) > 0:
        for col in num_cols:
            if df[col].isnull().sum() > 0:
                median_val = df[col].median()
                df[col] = df[col].fillna(median_val)
                report.append(f"✅ Imputed missing values in '{col}' with median.")

    cat_cols = df.select_dtypes(include=['object']).columns
    if len(cat_cols) > 0:
        df[cat_cols] = df[cat_cols].fillna("Unknown")
        
    return df, report
